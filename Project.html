# create a working raw copy (preserve data_file)
bliz_raw <- data_file

bliz_clean <- bliz_raw %>%

  # 1) Basic normalization and safe character extraction (all arguments inside mutate(), separated by commas)
  mutate(
    current_title = coalesce(as.character(current_title), "") %>% str_squish(),
    status = coalesce(as.character(status), "Unknown") %>% str_squish(),
    salary_type_orig = as.character(salary_type),
    current_salary_raw = as.character(current_salary),
    percent_incr_raw = as.character(percent_incr)
  ) %>%

  # 2) Parse numeric salary and percent fields (remove $ and commas)
  mutate(
    current_salary_num = current_salary_raw %>%
      str_remove_all("[$,]") %>%
      str_trim() %>%
      na_if("") %>%
      as.numeric(),

    percent_incr_num = percent_incr_raw %>%
      str_remove_all("[%,$]") %>%
      str_trim() %>%
      na_if("") %>%
      as.numeric()
  ) %>%

  # 3) Canonicalize salary_type into "hour"/"year" (case_when entries separated by commas)
  mutate(
    salary_type_clean = coalesce(salary_type_orig, "") %>%
      str_to_lower() %>%
      str_squish(),

    salary_type_clean = case_when(
      str_detect(salary_type_clean, "hour|hr|hourly") ~ "hour",
      str_detect(salary_type_clean, "year|yr|yearly|annual|salary") ~ "year",
      TRUE ~ NA_character_
    )
  ) %>%

  # 4) Compute annual_salary:
  # hourly -> numeric * (40 * 52), yearly -> numeric as-is
  mutate(
    annual_salary = case_when(
      salary_type_clean == "hour" & !is.na(current_salary_num) ~ current_salary_num * (40 * 52),
      salary_type_clean == "year" & !is.na(current_salary_num) ~ current_salary_num,
      TRUE ~ NA_real_
    )
  ) %>%

  # 5) Preserve optional fields (safe extraction)
  mutate(
    location = if ("location" %in% names(.)) coalesce(as.character(location), NA_character_) else NA_character_,
    performance_rating = if ("performance_rating" %in% names(.)) coalesce(as.character(performance_rating), NA_character_) else NA_character_
  )

bliz_raw <- data_file


bliz_clean <- bliz_raw %>%

  # 1) Basic normalization and keep raw text copies for debugging
  mutate(
    current_title = coalesce(as.character(current_title), "") %>% str_squish(),
    status = coalesce(as.character(status), "Unknown") %>% str_squish(),
    salary_type_orig = as.character(salary_type),         # original salary_type text kept
    current_salary_raw = as.character(current_salary),    # original salary string kept
    percent_incr_raw = as.character(percent_incr)        # original percent_incr string kept
  ) %>%

  # 2) Parse numeric salary and percent (strip $ , % etc.)
  mutate(
    current_salary_num = current_salary_raw %>%
      str_remove_all("[$,]") %>%    # remove dollar and comma
      str_trim() %>%
      na_if("") %>%
      as.numeric(),

    percent_incr_num = percent_incr_raw %>%
      str_remove_all("[%,$]") %>%
      str_trim() %>%
      na_if("") %>%
      as.numeric()
  ) %>%

  # 3) Canonicalize salary_type into "hour"/"year" using pattern matching
  mutate(
    salary_type_clean = coalesce(salary_type_orig, "") %>%
      str_to_lower() %>%
      str_squish(),

    salary_type_clean = case_when(
      str_detect(salary_type_clean, "hour|hr|hourly") ~ "hour",
      str_detect(salary_type_clean, "year|yr|yearly|annual|salary") ~ "year",
      TRUE ~ NA_character_
    )
  ) %>%

  # 4) Compute annual_salary (hour -> *40*52, year -> as-is). require numeric salary present.
  mutate(
    annual_salary = case_when(
      salary_type_clean == "hour" & !is.na(current_salary_num) ~ current_salary_num * (40 * 52),
      salary_type_clean == "year" & !is.na(current_salary_num) ~ current_salary_num,
      TRUE ~ NA_real_
    )
  ) %>%

  # 5) Create title_group from keywords, default to "Other", then lump to top 6
  mutate(
    title_lower = str_to_lower(current_title),
    title_group = case_when(
      str_detect(title_lower, "engineer") ~ "Engineer",
      str_detect(title_lower, "programmer|\\bdev\\b|developer|software") ~ "Engineer",
      str_detect(title_lower, "designer") ~ "Designer",
      str_detect(title_lower, "artist") ~ "Artist",
      str_detect(title_lower, "producer") ~ "Producer",
      str_detect(title_lower, "manager|lead|director") ~ "Management",
      str_detect(title_lower, "qa|quality") ~ "QA",
      str_detect(title_lower, "support|hr|people|recruit") ~ "Operations",
      TRUE ~ "Other"
    ),
    title_group = fct_lump_n(factor(title_group), n = 6, other_level = "Other")
  ) %>%

  # 6) Preserve optional columns (safe extraction), keep original raw strings for debug
  mutate(
    location = if ("location" %in% names(.)) coalesce(as.character(location), NA_character_) else NA_character_,
    performance_rating = if ("performance_rating" %in% names(.)) coalesce(as.character(performance_rating), NA_character_) else NA_character_
  )
